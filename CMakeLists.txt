cmake_minimum_required(VERSION 3.20)
project(graphnet_cpp VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(graphnet src/graphnet.cpp)
target_include_directories(graphnet PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_options(graphnet PRIVATE $<$<CXX_COMPILER_ID:GNU,Clang>:-Wall;-Wextra;-Wpedantic>)
add_executable(graphnet-demo src/main.cpp)
target_link_libraries(graphnet-demo PRIVATE graphnet)
enable_testing()
add_executable(graphnet-tests tests/graphnet_tests.cpp)
target_link_libraries(graphnet-tests PRIVATE graphnet)
add_test(NAME graphnet-tests COMMAND graphnet-tests)
